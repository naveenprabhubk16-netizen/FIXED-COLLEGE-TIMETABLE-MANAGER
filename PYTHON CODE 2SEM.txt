from flask import Flask, render_template, request

app = Flask(__name__)

# Fixed Timetable (Immutable using Tuple)
TIMETABLE = {
    "CSE": {
        "Monday": ("Maths","Physics","Chemistry","Lunch","English","CS","Library"),
        "Tuesday": ("CS","Maths","Physics","Lunch","Chemistry","English","Sports"),
        "Wednesday": ("Physics","CS","Maths","Lunch","English","Chemistry","Lab"),
        "Thursday": ("Maths","English","CS","Lunch","Physics","Chemistry","Library"),
        "Friday": ("Chemistry","Maths","CS","Lunch","Physics","English","Sports")
    }
}

# Home Page
@app.route("/")
def home():
    return render_template("home.html")

# Selection Page
@app.route("/select", methods=["GET", "POST"])
def select():
    subject = None

    if request.method == "POST":
        dept = request.form["dept"]
        day = request.form["day"]
        hour = int(request.form["hour"])

        subject = TIMETABLE[dept][day][hour]

    return render_template("select.html", subject=subject)

# Full Timetable Page
@app.route("/full/<dept>")
def full(dept):
    return render_template("full.html", timetable=TIMETABLE[dept], dept=dept)

# Run Server
if __name__ == "__main__":
    app.run(debug=True)
